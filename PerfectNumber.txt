Java Program to Check perfect Number

Solution:
import java.util.Scanner;

public class PerfectNumber {
    public static void main(String[] args) {
        // Create a Scanner object to take user input
        Scanner scanner = new Scanner(System.in);

        // Prompt the user to enter a number
        System.out.print("Enter a number: ");
        int num = scanner.nextInt();

        // Initialize sum to store the sum of divisors
        int sum = 0;

        // Loop to find the divisors of the number
        for (int i = 1; i <= num / 2; i++) {
            if (num % i == 0) {  // If 'i' divides 'num' perfectly
                sum += i;  // Add divisor to sum
            }
        }

        // Check if sum of divisors equals the original number
        if (sum == num && num != 0) {
            System.out.println(num + " is a Perfect Number");
        } else {
            System.out.println(num + " is not a Perfect Number");
        }

        // Close the scanner object
        scanner.close();
    }
}
