import java.util.Scanner;

public class PalindromeNumber {
    public static void main(String[] args) {
        // Create a scanner object to read input from the user
        Scanner scanner = new Scanner(System.in);

        // Ask the user for input
        System.out.print("Enter a number: ");
        
        // Read the integer input from the user
        int originalNumber = scanner.nextInt();
        
        // Initialize a variable to store the reversed number
        int reversedNumber = 0;
        
        // Store the original number in a temporary variable for comparison
        int number = originalNumber;

        // Reverse the number using a while loop
        while (number != 0) {
            // Extract the last digit of the number
            int digit = number % 10;
            
            // Build the reversed number
            reversedNumber = reversedNumber * 10 + digit;
            
            // Remove the last digit from the number
            number = number / 10;
        }
        
        // Check if the original number is equal to the reversed number
        if (originalNumber == reversedNumber) {
            // If the number is equal to its reverse, it's a palindrome
            System.out.println(originalNumber + " is a palindrome.");
        } else {
            // Otherwise, it's not a palindrome
            System.out.println(originalNumber + " is not a palindrome.");
        }

        // Close the scanner object
        scanner.close();
    }
}

import java.util.Scanner;

public class PalindromeNumber {
    
    // Recursive function to check if the number is a palindrome
    public static boolean isPalindrome(int number, int temp) {
        // Base case: If the number is 0, all digits have been checked
        if (number == 0) {
            return temp == 0;  // If the reversed number matches the original, it's a palindrome
        }

        // Extract the last digit of the number
        int digit = number % 10;
        
        // Reverse the number by constructing the number from the last digit
        temp = temp * 10 + digit;

        // Recursive call with the remaining digits of the number
        return isPalindrome(number / 10, temp);
    }

    public static void main(String[] args) {
        // Create a scanner object to take input from the user
        Scanner scanner = new Scanner(System.in);

        // Ask the user for input
        System.out.print("Enter a number: ");
        
        // Read the integer input from the user
        int originalNumber = scanner.nextInt();

        // Call the recursive function to check if the number is a palindrome
        if (isPalindrome(originalNumber, 0)) {
            System.out.println(originalNumber + " is a palindrome.");
        } else {
            System.out.println(originalNumber + " is not a palindrome.");
        }

        // Close the scanner object
        scanner.close();
    }
}
