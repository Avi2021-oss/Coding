import java.util.Scanner; // Import Scanner class for user input

public class Fibonacci {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);  // Create a Scanner object to read input from the user
        
        System.out.print("Enter the number of terms: ");
        int n = sc.nextInt();  // Read the number of terms for Fibonacci sequence
        
        int first = 0, second = 1;  // Initialize the first two terms of the Fibonacci sequence
        System.out.print("Fibonacci Series: ");
        
        // Print the Fibonacci series
        for (int i = 1; i <= n; i++) {
            System.out.print(first + " ");  // Print the current Fibonacci number
            int nextTerm = first + second;  // Calculate the next term in the series
            first = second;  // Update first to be the second term
            second = nextTerm;  // Update second to be the newly calculated term
        }
        sc.close();  // Close the Scanner object to avoid resource leak
    }
}
